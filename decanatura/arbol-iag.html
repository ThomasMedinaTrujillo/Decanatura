<script>

  const S = { step: 1, answers: [] };


  const Q1 = '¿Hay competencias en esta evaluación que el estudiante debe demostrar de forma completamente autónoma —sin apoyo externo de ningún tipo— porque son fundamento para lo que viene después, en el programa o en su vida profesional?';

  const Q2 = '¿Puede garantizar un entorno controlado o supervisado donde efectivamente no se utilice IA?';


  const LEVELS = [

    { n:1, cls:'n1', txt:'Quiero evaluar habilidades esenciales sin IAG.', href:'#nivel-1' },

    { n:2, cls:'',   txt:'Quiero que la IAG sea solo una herramienta de apoyo, para la planeación e ideación previas a la tarea a evaluar.', href:'#nivel-2' },

    { n:3, cls:'',   txt:'Quiero que la IAG participe en la elaboración de la tarea, y haya un foco en el juicio crítico de lo que genera la IA.', href:'#nivel-3' },

    { n:4, cls:'',   txt:'Quiero que el uso crítico y estratégico de la IAG sea parte explícita de lo que se evalúa.', href:'#nivel-4' },

    { n:5, cls:'',   txt:'Quiero que la innovación, creación y exploración de nuevas posibilidades con la IAG sea parte central de la tarea.', href:'#nivel-5' },

  ];


  function levelCards(include) {

    return include.map(n => {

      const l = LEVELS[n-1];

      return `<a class="level-card" href="${l.href}">

        <div class="level-num ${l.cls}">N${l.n}</div>

        <div class="level-text">${l.txt}</div>

        <div class="level-arrow">→</div>

      </a>`;

    }).join('');

  }


  function references() {

    return `<div class="ref-section">

      <div class="ref-label">Referencia</div>

      <div class="ref-text">Perkins, M., Roe, J., &amp; Furze, L. (2025). Reimagining the Artificial Intelligence Assessment Scale (AIAS): A refined framework for educational assessment. <em>Journal of University Teaching and Learning Practice, 22</em>(7). <a href="https://doi.org/10.53761/rrm4y757" target="_blank">https://doi.org/10.53761/rrm4y757</a></div>

      <div class="ref-adapt">Adaptado por Laura Sofía Gutiérrez.</div>

    </div>`;

  }


  function render() {

    const bd = document.getElementById('bd');

    bd.innerHTML = '';


    S.answers.forEach((ans, i) => {

      const qText = i === 0 ? Q1 : Q2;

      const bcls = ans==='SÍ' ? 'badge badge-si' : ans==='NO' ? 'badge badge-no' : 'badge badge-parc';

      const h = document.createElement('div');

      h.className = 'hist';

      h.innerHTML = `<div class="hist-q"><strong>Pregunta ${i+1}:</strong> ${qText}</div><span class="${bcls}">${ans}</span>`;

      bd.appendChild(h);


      if (i===0 && ans==='PARCIALMENTE') {

        const note = document.createElement('div');

        note.className = 'note';

        note.innerHTML = '💡 <strong>Nota:</strong> Considera <strong>dividir la evaluación</strong>: una parte sin IAG (Nivel 1) para las competencias que requieren autonomía total, y otra donde sí la permites (Niveles 2–5) para el resto. Con eso en mente, avanza a la siguiente pregunta.';

        bd.appendChild(note);

      }

    });


    if      (S.step===1)          renderQ1(bd);

    else if (S.step===2)          renderQ2(bd);

    else if (S.step==='r-25')     renderR25(bd);

    else if (S.step==='r-n1')     renderRN1(bd);

    else if (S.step==='r-rediseno') renderRRediseno(bd);


    updateDots();

  }


  function renderQ1(bd) {

    const d = document.createElement('div');

    d.className = 'q-card';

    d.innerHTML = `

      <div class="q-num">Pregunta 1 de 2</div>

      <div class="q-text">${Q1}</div>

      <div class="btns">

        <button class="btn btn-si"   onclick="a1('SÍ')">Sí</button>

        <button class="btn btn-no"   onclick="a1('NO')">No</button>

        <button class="btn btn-parc" onclick="a1('PARCIALMENTE')">Parcialmente</button>

      </div>

      ${references()}`;

    bd.appendChild(d);

  }


  function renderQ2(bd) {

    const d = document.createElement('div');

    d.className = 'q-card';

    d.innerHTML = `

      <div class="q-num">Pregunta 2 de 2</div>

      <div class="q-text">${Q2}</div>

      <div class="btns">

        <button class="btn btn-si" onclick="a2('SÍ')">Sí</button>

        <button class="btn btn-no" onclick="a2('NO')">No</button>

      </div>

      ${references()}`;

    bd.appendChild(d);

  }


  function renderR25(bd) {

    const d = document.createElement('div');

    d.className = 'res-card';

    d.innerHTML = `

      <div class="res-pill pill-yellow">Resultado</div>

      <div class="res-title">Los Niveles 2 al 5 son el territorio pertinente para su evaluación</div>

      <div class="res-body">Dado que no hay competencias que exijan demostración completamente autónoma, puede diseñar evaluaciones donde la IAG tenga un rol activo y pedagógicamente justificado. ¿Cuánta participación quiere darle? Elija la afirmación que mejor describe su intención:</div>

      <div class="levels-intro">Ir directamente al nivel que busco</div>

      <div class="level-cards">${levelCards([2,3,4,5])}</div>

      <hr class="sep">

      <button class="restart" onclick="restart()">↺ Volver a empezar</button>

      ${references()}`;

    bd.appendChild(d);

  }


  function renderRN1(bd) {

    const d = document.createElement('div');

    d.className = 'res-card';

    d.innerHTML = `

      <div class="res-pill pill-blue">Nivel 1 · Sin IAG</div>

      <div class="res-title">Su evaluación se ubica en el Nivel 1: Sin uso de IAG</div>

      <div class="res-body">Hay competencias que requieren demostración autónoma <em>y</em> puede garantizar las condiciones para verificarlo. Puede declarar su evaluación libre de IAG con respaldo pedagógico sólido.</div>

      <div class="res-links">

        <a class="res-link link-pri" href="#nivel-1">Ver Nivel 1 en detalle →</a>

      </div>

      <hr class="sep">

      <button class="restart" onclick="restart()">↺ Volver a empezar</button>

      ${references()}`;

    bd.appendChild(d);

  }


  function renderRRediseno(bd) {

    const d = document.createElement('div');

    d.className = 'res-card';

    d.innerHTML = `

      <div class="res-pill pill-action">Acción requerida</div>

      <div class="res-title">Su evaluación necesita ser rediseñada</div>

      <div class="res-body">Declarar "sin IAG" sin poder verificarlo no protege el aprendizaje. Tiene dos caminos: rediseñar las condiciones de aplicación para garantizar el entorno controlado, o replantear la evaluación para que la IAG tenga un rol deliberado y con propósito. ¿Qué dirección se ajusta más a lo que busca?</div>

      <div class="levels-intro">Explorar los niveles del AIAS</div>

      <div class="level-cards">${levelCards([1,2,3,4,5])}</div>

      <hr class="sep">

      <button class="restart" onclick="restart()">↺ Volver a empezar</button>

      ${references()}`;

    bd.appendChild(d);

  }


  function a1(ans) {

    S.answers = [ans];

    S.step = ans==='NO' ? 'r-25' : 2;

    render();

  }


  function a2(ans) {

    S.answers.push(ans);

    S.step = ans==='SÍ' ? 'r-n1' : 'r-rediseno';

    render();

  }


  function restart() { S.step=1; S.answers=[]; render(); }


  function updateDots() {

    const d1=document.getElementById('d1'), d2=document.getElementById('d2');

    if      (S.step===1) { d1.className='dot active'; d2.className='dot'; }

    else if (S.step===2) { d1.className='dot done';   d2.className='dot active'; }

    else                 { d1.className='dot done';   d2.className='dot done'; }

  }


  render();

</script>
